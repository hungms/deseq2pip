####################### PKGDOWN Workflows #######################
## 1. rename pkgdown.yaml to just pkgdown

## 2. run pkgdown workflow only after R-CMD-check is successful

>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>
# Workflow derived from https://github.com/r-lib/actions/tree/v2/examples
# Need help debugging build failures? Start at https://github.com/r-lib/actions#where-to-find-help
on:
  workflow_run:
    workflows: ["R-CMD-check"]
    branches: [main, master]
    types:
      - completed
  release:
    types: [published]
  workflow_dispatch:

name: pkgdown

permissions: read-all

jobs:
  pkgdown:
    runs-on: ubuntu-latest
    # Only run if the R-CMD-check workflow was successful
    if: ${{ github.event.workflow_run.conclusion == 'success' || github.event_name == 'release' || github.event_name == 'workflow_dispatch' }}
    # Only restrict concurrency for non-PR jobs
    concurrency:
      group: pkgdown-${{ github.run_id }}

log_output <- function(call, save_dir, expr = NULL) {
  # ... (same as before for log_vec and log_file) ...

  writeLines(log_vec, log_file)

  if (!is.null(expr)) {
    sink(log_file, append = TRUE, split = TRUE)
    sink(log_file, append = TRUE, split = TRUE, type = "message")
    on.exit({
      sink(type = "message")
      sink()
    }, add = TRUE)
    withCallingHandlers(
      eval(expr),
      warning = function(w) {
        message("WARNING: ", conditionMessage(w))
        invokeRestart("muffleWarning")
      }
    )
    # If an error occurs, it will stop here and propagate up
  }
}
